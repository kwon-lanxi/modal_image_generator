HERE = Path(__file__).parent
image = modal.Image.debian_slim(python_version="3.12").pip_install_from_requirements(str(HERE/"requirements.txt")).add_local_file(str(HERE/"logic.py"), "/root/logic.py", copy=True).add_local_file(str(HERE/"mini.html"), "/root/mini.html", copy=True)
app = modal.App("krea2-telegram")

credentials = [modal.Secret.from_name("tg-token-image"), modal.Secret.from_name("krea2-telegram-config"), modal.Secret.from_name("krea2-api")]

@app.cls(image=image, volumes={"/state":volume}, cpu=.125, memory=256, max_containers=1, scaledown_window=2)
@modal.concurrent(max_inputs=1)
class State:
    @modal.method()
    def call(self, op, key="", value=None):
        volume.reload()
        with closing(sqlite3.connect("/state/bot.sqlite")) as db, db:
            db.execute("CREATE TABLE IF NOT EXISTS kv (k TEXT PRIMARY KEY, v TEXT NOT NULL)")
            if op=="get":
                row=db.execute("SELECT v FROM kv WHERE k=?",(key,)).fetchone();result=json.loads(row[0]) if row else value
            elif op=="put":
                db.execute("INSERT OR REPLACE INTO kv VALUES (?,?)",(key,json.dumps(value)));result=value
            elif op=="insert":
                result=db.execute("INSERT OR IGNORE INTO kv VALUES (?,?)",(key,json.dumps(value))).rowcount==1
            elif op=="alias":
                row=db.execute("SELECT v FROM kv WHERE k='aliases'").fetchone()
                aliases=json.loads(row[0]) if row else {}
                old=aliases.get(key)
                if old and old["id"]!=value["id"]:raise ValueError("Alias existente para otro archivo")
                if len(aliases)>=1000 and key not in aliases:raise ValueError("Máximo1000 alias")
                aliases[key]=value
                db.execute("INSERT OR REPLACE INTO kv VALUES ('aliases',?)",(json.dumps(aliases),));result=aliases
            elif op=="list": result={k:json.loads(v) for k,v in db.execute("SELECT k,v FROM kv WHERE k LIKE ? ORDER BY rowid DESC LIMIT 100",(key+"%",))}
            elif op=="active":
                statuses=("new",) if key=="update:" else ("ready","running","result","submitting","submission_unknown")
                placeholders=",".join("?" for _ in statuses)
                result={k:json.loads(v) for k,v in db.execute("SELECT k,v FROM kv WHERE k LIKE ? AND json_extract(v,'$.status') IN ("+placeholders+")",(key+"%",*statuses))}
            else: raise ValueError("Unknown state operation")
        volume.commit()
        return result


def state(op,key="",value=None): return modal.Cls.from_name("krea2-telegram","State")().call.remote(op,key,value)


def api(method,path,**kwargs):
    import requests
    r=requests.request(method,os.environ["KREA2_URL"].rstrip("/")+path,
        headers={"Authorization":"Bearer "+os.environ["KREA2_API_TOKEN"],**kwargs.pop("headers",{})},timeout=(10,60),**kwargs)
    if r.status_code>=400: raise RuntimeError("Krea API HTTP "+str(r.status_code))
    return r


def tg(method, **data):
    import requests
    # Never print request URLs/exceptions: the Bot API embeds its token in the path.
    if time.time()<state("get","telegram_not_before",0):raise RuntimeError("Telegram rate limit active")
    files=data.pop("_files",None)
    try:
        r=requests.post("https://api.telegram.org/bot"+os.environ["TELEGRAM_KREA2_BOT_TOKEN"]+"/"+method,
                        data=data if files else None,json=None if files else data,files=files,timeout=(10,60))
        result=r.json()
    except Exception: raise RuntimeError("Telegram transport error; delivery may be ambiguous") from None
    if not result.get("ok"):
        if result.get("error_code")==429:
            state("put","telegram_not_before",time.time()+int(result.get("parameters",{}).get("retry_after",1)))
            raise RuntimeError("Telegram rate limit; retry after "+str(result.get("parameters",{}).get("retry_after",1)))
        if "message is not modified" in result.get("description",""): return None
        raise RuntimeError("Telegram API error "+str(result.get("error_code")))
    return result.get("result")


def buttons(rows): return {"inline_keyboard":rows}
def button(text,data): return {"text":text,"callback_data":data}
def library(): return buttons([[{"text":"Biblioteca y galería", "web_app":{"url":os.environ["BOT_PUBLIC_URL"].rstrip("/")+"/mini"}}]])
def send(chat,text,markup=None): return tg("sendMessage",chat_id=chat,text=text[:4096],**({"reply_markup":markup} if markup else {}))


def draft(chat,recipe):
    key=uuid.uuid4().hex[:16]
    state("put","job:"+key,{"status":"draft","chat":chat,"recipe":recipe})
    text=recipe["positive_prompt"]+"\n\nLoRAs: "+json.dumps(recipe.get("lora",{}),ensure_ascii=False)+f"\n{recipe['width']}×{recipe['height']} de muestreo (VAE2x puede duplicar salida)"
    send(chat,text,buttons([[button("Generar","go:"+key),button("Descartar","drop:"+key)]]))


def handle(update):
    from logic import parse_prompt
    callback=update.get("callback_query")
    if callback:
        try: tg("answerCallbackQuery",callback_query_id=callback["id"])
        except RuntimeError: pass
        action,_,key=callback.get("data","").partition(":")
        job=state("get","job:"+key)
        if not job: return
        if action=="go" and job["status"]=="draft":
            job["status"]="ready";state("put","job:"+key,job)
        elif action=="drop" and job["status"]=="draft":
            job["status"]="discarded";state("put","job:"+key,job)
        elif action in {"repeat","vary"} and job["status"] in {"done","delivery_unknown"}:
            recipe=dict(job["recipe"])
            recipe["seed"]=job.get("result",{}).get("seed",recipe.get("seed",0)) if action=="repeat" else secrets.randbelow(2**63-1)+1
            draft(job["chat"],recipe)
        elif action=="recover" and job["status"]=="needs_review":
            job["status"]="submission_unknown";state("put","job:"+key,job)
        elif action=="deliver" and job.get("result"):
            job["status"]="result";state("put","job:"+key,job)
        return
    msg=update.get("message",{});chat=msg["chat"]["id"];text=msg.get("text","")
    if text.startswith(("/start","/help","/loras","/galeria")):
        send(chat,"Krea2 personal\nPrompt @alias:0.8 --size vertical\n-w 896 -h 1152 --seed 42 --steps 8 --cfg 1\n/descargar ENLACE nombre\nLos alias y pesos se editan en la biblioteca. Cada generación pide confirmación.",library());return
    if text.startswith("/trabajos"):
        jobs=state("list","job:")
        lines=[k[4:]+" · "+v["status"] for k,v in list(jobs.items())[-10:]]
        send(chat,"Últimos trabajos\n"+"\n".join(lines),library());return
    if text.startswith("/descargar "):
        import shlex
        words=shlex.split(text)
        if not 2<=len(words)<=3: raise ValueError("/descargar ENLACE [nombre]")
        data={"url":words[1],"target":"lora"}
        if len(words)==3:data["name"]=words[2]
        key=uuid.uuid4().hex[:16]
        state("put","job:"+key,{"status":"ready","chat":chat,"operation":"download","recipe":data});return
    if text.startswith("/generar "):text=text[len("/generar "):]
    draft(chat,parse_prompt(text,state("get","aliases",{})))


def advance(key,job):
    from logic import progress_text
    if job["status"]=="ready":
        job["status"]="submitting";job["request_id"]=hashlib.sha256(("tg-krea-"+key).encode()).hexdigest()[:32]
        job["message_id"]=send(job["chat"],"Trabajo aceptado. Preparando…")["message_id"]
        state("put","job:"+key,job)
        operation=job.get("operation","generate")
        route="/download" if operation=="download" else "/jobs"
        payload=job["recipe"] if operation=="download" else {"operation":"generate",**job["recipe"]}
        try:
            accepted=api("POST",route,json=payload,headers={"Idempotency-Key":"tg-krea-"+key}).json()
            job.update(status="running",status_url=accepted["status_url"])
        except Exception: job["status"]="submission_unknown"
        state("put","job:"+key,job)
    if job["status"] in {"submitting","submission_unknown"}:
        # A lost POST response must never cause an automatic second generation.
        try:r=api("GET","/records/"+job["request_id"]).json()
        except Exception:
            job["status"]="needs_review";state("put","job:"+key,job)
            tg("editMessageText",chat_id=job["chat"],message_id=job["message_id"],text="Respuesta de envío incierta. No volveré a generar automáticamente. Podés consultar de nuevo sin GPU.",reply_markup=buttons([[button("Consultar trabajo","recover:"+key)]]))
            return
        if r.get("status")=="done":
            results=api("GET","/outputs?limit=500").json().get("outputs",[])
            result=next((x for x in results if x.get("task_id")==job["request_id"]),None)
            if result:job.update(status="result",result=result);state("put","job:"+key,job)
        return
    if job["status"]=="running":
        response=api("GET",job["status_url"])
        if response.status_code==202:
            event=api("GET","/records/"+job["request_id"]+"/progress").json().get("progress") or {}
            text=progress_text(event)
            if text!=job.get("last_text"):
                tg("editMessageText",chat_id=job["chat"],message_id=job["message_id"],text=text)
                job["last_text"]=text;state("put","job:"+key,job)
            return
        result=response.json()
        if not result.get("success"):
            job["status"]="failed";state("put","job:"+key,job);send(job["chat"],"El trabajo falló. Consultá sus logs: "+job["request_id"]);return
        job.update(status="result",result=result);state("put","job:"+key,job)
    if job["status"]=="result":
        result=job["result"]
        job["status"]="delivery_unknown";state("put","job:"+key,job)
        if result.get("images"):
            from urllib.parse import quote
            im=result["images"][0]
            raw=api("GET","/serve_image?filename="+quote(im["filename"],safe="")).content
            markup=buttons([[button("Repetir","repeat:"+key),button("Nueva variante","vary:"+key)], [button("Reenviar original sin GPU","deliver:"+key)]])
            tg("sendDocument",chat_id=job["chat"],caption="Krea2 · seed "+str(result.get("seed_string",result.get("seed")))+" · "+str(result.get("created_at","")),reply_markup=json.dumps(markup),_files={"document":(im["filename"],raw,"image/png")})
        else: send(job["chat"],"Descarga terminada. Abrí la biblioteca para asignar alias.",library())
        job["status"]="done";state("put","job:"+key,job)
        tg("editMessageText",chat_id=job["chat"],message_id=job["message_id"],text="Listo · guardado en el servidor",reply_markup=buttons([[button("Reenviar","deliver:"+key)]]))


@app.function(image=image,secrets=credentials,cpu=.125,memory=256,max_containers=1,timeout=120,scaledown_window=2,schedule=modal.Period(minutes=1),retries=0)
def process():
    until=time.monotonic()+45
    while True:
        for k,entry in state("active","update:").items():
            if entry["status"]!="new":continue
            entry["status"]="handling";state("put",k,entry)
            try:handle(entry["update"])
            except Exception:
                msg=entry["update"].get("message",{})
                if msg.get("chat"):send(msg["chat"]["id"],"No pude procesarlo. Revisá alias, pesos y parámetros; /help")
            entry["status"]="done";state("put",k,entry)
        active=False
        for full,job in state("active","job:").items():
            if job["status"] not in {"ready","running","result","submitting","submission_unknown"}:continue
            active=True
            try:advance(full[4:],job)
            except Exception as exc:
                print(json.dumps({"event":"bot.job.error","job_id":full[4:],"error":type(exc).__name__}))
        if not active or time.monotonic()>until:return
        time.sleep(5)


@app.function(image=image,secrets=credentials,cpu=.125,memory=256,max_containers=1,scaledown_window=2)
@modal.concurrent(max_inputs=16)
@modal.asgi_app()
def web():
    from fastapi import FastAPI,Request,HTTPException
    from fastapi.responses import HTMLResponse,Response
    from logic import OWNER,validate_init
    api_app=FastAPI(docs_url=None,redoc_url=None,openapi_url=None)
    def auth(request):
        try:validate_init(request.headers.get("X-Telegram-Init-Data",""),os.environ["TELEGRAM_KREA2_BOT_TOKEN"])
        except Exception:raise HTTPException(401,"Reabrí la Mini App desde Telegram")
    @api_app.get("/mini")
    def mini():return HTMLResponse(Path("/root/mini.html").read_text())
    @api_app.post("/webhook")
    async def webhook(request:Request):
        if not hmac.compare_digest(request.headers.get("X-Telegram-Bot-Api-Secret-Token", ""),os.environ["TELEGRAM_WEBHOOK_SECRET"]):raise HTTPException(403)
        raw=await request.body()
        if len(raw)>65536:raise HTTPException(413)
        try:u=json.loads(raw);assert isinstance(u["update_id"],int)
        except Exception:raise HTTPException(400)
        m=u.get("message") or u.get("callback_query",{}).get("message",{})
        sender=u.get("callback_query",{}).get("from") or u.get("message",{}).get("from",{})
        if sender.get("id")!=OWNER or m.get("chat",{}).get("type")!="private":return {"ok":True}
        inserted=await State().call.remote.aio("insert","update:"+str(u["update_id"]),{"status":"new","update":u})
        # Spawn is acknowledged only after persistence. Periodic recovery covers a lost spawn.
        if inserted:await process.spawn.aio()
        return {"ok":True}
    @api_app.get("/mini/loras")
    def loras(request:Request,q:str="",page:int=0):
        auth(request)
        entries=api("GET","/list_models?type=lora").json()["models"]
        entries=[x for x in entries if q.lower() in (x["id"]+" "+x["name"]).lower()]
        page=max(0,page);return {"items":entries[page*20:(page+1)*20],"total":len(entries)}
    @api_app.get("/mini/aliases")
    def aliases(request:Request):auth(request);return state("get","aliases",{})
    @api_app.post("/mini/aliases")
    async def alias(request:Request):
        import re,math
        auth(request);raw=await request.body()
        if len(raw)>4096:raise HTTPException(413)
        data=json.loads(raw);name=data.get("alias","");weight=float(data.get("weight",.8))
        if not re.fullmatch(r"[A-Za-z0-9_-]{1,32}",name) or not math.isfinite(weight) or abs(weight)>2:raise HTTPException(400,"Alias/peso inválido")
        entries=await __import__('asyncio').to_thread(lambda:api("GET","/list_models?type=lora").json()["models"])
        if data.get("id") not in {x["id"] for x in entries}:raise HTTPException(400,"Archivo no encontrado")
        # Serialized read/merge happens inside State (not in concurrent HTTP handlers).
        return await State().call.remote.aio("alias",name,{"id":data["id"],"weight":weight})
    @api_app.get("/mini/history")
    def history(request:Request):auth(request);return api("GET","/outputs?limit=100").json()
    @api_app.get("/mini/image")
    def original(request:Request,filename:str):
        from urllib.parse import quote
        auth(request);r=api("GET","/serve_image?filename="+quote(filename,safe=""));return Response(r.content,media_type=r.headers.get("content-type","image/png"))
    return api_app


@app.local_entrypoint()
def activate(url:str):
    # Explicit activation; deployment alone never replaces an existing Telegram webhook.
    if not url.startswith("https://"):raise ValueError("HTTPS required")
    configure.remote(url)

@app.function(image=image,secrets=credentials,cpu=.125,memory=256)
def configure(url):
    me=tg("getMe")
    result=tg("setWebhook",url=url.rstrip("/")+"/webhook",secret_token=os.environ["TELEGRAM_WEBHOOK_SECRET"],allowed_updates=["message","callback_query"],max_connections=2,drop_pending_updates=False)
    print(json.dumps({"username":me["username"],"webhook_set":result}))


@app.function(image=image,secrets=credentials,cpu=.125,memory=256)
def diagnostics():
    me=tg("getMe");info=tg("getWebhookInfo")
    return {"username":me["username"],"webhook_url":info.get("url"),"pending_updates":info.get("pending_update_count"),
            "has_delivery_error":bool(info.get("last_error_date")),"state_ok":isinstance(state("get","aliases",{}),dict),
            "krea_health":api("GET","/health").json()}
